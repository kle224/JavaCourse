public class AssignmentGroup {
	private final String sysID;		// sys_id of the assignment group in SN
	private final String name;		// optional display name

	// creates an assignment group with only a sys_id
	public AssignmentGroup(String sysID) {
		this(sysID, null);
	}
	
	// creates an assignment group with sys_id and display name
	public AssignmentGroup(String sysID, String name) {
		if (sysID == null || sysID.isBlank()) {		// allows creation only if a sys_id is known
			throw new IllegalArgumentException("AssignmentGroup sysID must not be null or empty");
		}
		
		this.sysID = sysID;
		this.name = name;
	}
	
	public String getSysID() {
		return sysID;
	}
	
	public String getName() {
		return name;
	}
	
	// returns a human-readable string to represent the assignment group
	@Override
	public String toString() {
		return name != null
				? "AssignmentGroup{name='" + name + "', sysID='" + sysID + "'}"
				: "AssignmentGroup{sysID='" + sysID + "'}";
	}
}

package de.schoenn.ServiceNow4J;

import java.io.ByteArrayOutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.lang.reflect.Array;

public class IOUtils{

	/**
	 * Gets the contents of an {@link InputStream} as a <code>byte[]</code>.
	 * <p>
	 * This method buffers the input internally, so there is no need to use a
	 * {@link BufferedInputStream}.
	 * 
	 * @param input the {@link InputStream} to read from
	 * @return the requested byte array
	 * @throws IOException if an I/O error occurs
	 */
	public static byte[] toByteArray(InputStream input) throws IOException{
		ByteArrayOutputStream output = new ByteArrayOutputStream();
		byte[] buffer = new byte[4096];
		int n = 0;
		while(-1 != (n = input.read(buffer))){
			output.write(buffer, 0, n);
		}
		return output.toByteArray();
	}
	
	public static <T> T concatArrays(T array1, T array2) {
		if (!array1.getClass().isArray() || !array2.getClass().isArray()) {
			throw new IllegalArgumentException("Only arrays are accepted.");
		}
		
		Class<?> compType = array1.getClass().getComponentType();
		if (!compType.equals(array2.getClass().getComponentType())) {
			throw new IllegalArgumentException("Arrays have to be of same type.");
		}
		
		int len1 = Array.getLength(array1);
		int len2 = Array.getLength(array2);
		
		@SuppressWarnings("unchecked")
		T result = (T) Array.newInstance(compType, len1 + len2);
		
		System.arraycopy(array1, 0, result, 0, len1);
		System.arraycopy(array2, 0, result, len1, len2);
		
		return result;
	}
}

		// Create a trust manager that does not validate certificate chains
        TrustManager[] trustAllCerts = new TrustManager[] {(TrustManager) new X509TrustManager() {
                public java.security.cert.X509Certificate[] getAcceptedIssuers() {
                    return null;
                }
                public void checkClientTrusted(X509Certificate[] certs, String authType) throws CertificateException {

                public void checkServerTrusted(X509Certificate[] certs, String authType) throws CertificateException {
                }
            }
        };
 
        // Install the all-trusting trust manager
        SSLContext sc = null;
		try {
			sc = SSLContext.getInstance("SSL");
		} catch (NoSuchAlgorithmException e) {
			e.printStackTrace();
		}
        try {
			sc.init(null, trustAllCerts, new java.security.SecureRandom());
		} catch (KeyManagementException e) {
			e.printStackTrace();
		}
        HttpsURLConnection.setDefaultSSLSocketFactory(sc.getSocketFactory());
 
        // Create all-trusting host name verifier
        HostnameVerifier allHostsValid = new HostnameVerifier() {
            public boolean verify(String hostname, SSLSession session) {
                return true;
            }
        };
 
        // Use the all-trusting host verifier
        HttpsURLConnection.setDefaultHostnameVerifier(allHostsValid);
	}
	
	// retrieves an incident from SN by its sys_id
	public Incident getIncident(String sysID) throws APIException {
		// creates an empty Incident object (acts as a data container for the API response)
		Incident incident = new Incident(sysID);
		
		try {
			// initialize the HTTP client (prepare the REST connection to the SN Table API)
			ServiceNowHttpsClient https = new ServiceNowHttpsClient(new URI(
										  endpoint + "/api/now/table/incident/" + sysID),
										  user, pass);

			// execute the API call and receive the response
			JSONObject response = https.getJSON();						// send the HTTP request and parse the JSON response
			// extract the relevant data object
			JSONObject result = response.getJSONObject("result");		// SN wraps the actual record inside the result object
			
			// map JSON fields to the incident object
			incident.setIncNumber(result.optString("incNumber", null));
			incident.setState(result.optString("state", null));
			incident.setActive(result.optBoolean("active", false));
			
			System.out.println("Result: " + result.toString(2));
			
			
			
			if (result.optJSONObject("cmdb_ci") != null) {
				incident.setCmdbCiSysID(result.getJSONObject("cmdb_ci").getString("value"));

		} catch (IOException e) {			// handle technical I/O errors
			e.printStackTrace();
		} catch (URISyntaxException e) {		// handle invalid URI syntax errors
			e.printStackTrace();
		}
		
		return incident;
	}
	
	// creates a new incident in SN
	// builds a JSON payload with the required incident data,
	// sends it to the SN Table API and maps the response
	// to a newly created Incident object
	public Incident createIncident(
			String shortDescription,
			String description,
			ConfigurationItem ci,
			AssignmentGroup group,
			Impact impact,
			Urgency urgency,
			String caller) throws APIException{
		// initialize the return object
		Incident incident = null;
		
		try {								
			// build the JSON payload for the API request (prepare all required fields for incident creation)			
			JSONObject payload = new JSONObject();
			payload.put("short_description", shortDescription);
			payload.put("description", description);
			payload.put("cmdb_ci", ci.getSysID());
			payload.put("contact_type", "u_interface");
			
			if (group != null) {
				payload.put("assignment_group", group.getSysID());
			}

			payload.put("impact", impact.getID());
			payload.put("urgency", urgency.getID());
			payload.put("caller_id", caller);
			
			// initialize the HTTP client for the POST request (prepare the REST connection to create a new incident)
			ServiceNowHttpsClient https =
					new ServiceNowHttpsClient(
							new URI(endpoint + "/api/now/table/incident"),
							user,
							pass,
							HTTPMethod.POST,
							new java.util.HashMap<>(),
							payload);
			
			// execute the API call and receive the response
			JSONObject response = https.getJSON();						// send the incident data to SN and parse the response
			// extract the created incident data
			JSONObject result = response.getJSONObject("result");		// SN returns the created record inside the result object
			
			// map JSON fields to the Incident object 
			incident = new Incident(
					result.getString("sys_id"),
					result.getString("number"));			
		} catch (IOException e){					// handle technical I/O errors
			e.printStackTrace();
		} catch (URISyntaxException e){			// handle invalid URI syntax errors
			e.printStackTrace();
		}
		
		return incident;
	}

package de.schoenn.ServiceNow4J;

import java.io.IOException;
import java.io.OutputStreamWriter;
import java.net.MalformedURLException;
import java.net.URI;
import java.net.URISyntaxException;
import java.net.URL;
import java.net.URLEncoder;
import java.nio.charset.StandardCharsets;
import java.util.Base64;
import java.util.HashMap;
import java.util.Map;

import javax.net.ssl.HttpsURLConnection;

import org.json.JSONObject;

public class ServiceNowHttpsClient{

	private final URL					url;
	private final String				user, pass;
	private final HTTPMethod			method;
	private final Map<String, String>	args;
	private final JSONObject			payload;
	private int							responseCode	= -1;
	private byte[]						content;

	// convenience constructor (default get) (allows simple GET requests without specifying method/args/payload)
	public ServiceNowHttpsClient(URI uri, String user, String pass) throws IOException {
		this(uri, user, pass, HTTPMethod.GET, new HashMap<String, String>());
	}

	// convenience constructor (method + args, empty payload) (allows requests with query parameters but without a JSON body)
	public ServiceNowHttpsClient(URI uri, String user, String pass, HTTPMethod method, Map<String, String> args) throws IOException {
		this(uri, user, pass, method, args, new JSONObject());
	}
	
	// full constructor
	// stores all required request data (URL, credentials, method, query args and payload)
	public ServiceNowHttpsClient(URI uri, String user, String pass, HTTPMethod method, Map<String, String> args, JSONObject payload) throws IOException {
		this.url = uri.toURL();
		this.user = user;
		this.pass = pass;
		this.method = method;
		this.args = args;
		this.payload = payload;
	}

	// core request execution method (builds final URL (incl. query parameters), sends the HTTPS request
	// and reads the response (or error) content
	private void getContent() throws IOException, APIException {
		URL url = this.url;				// build final URL (SN requests require angular=true + optional query args)
//		args.put("angular", "true");	// ensure args map exists and add required SN parameter
		
		// build query string safely (no substring issues if args is empty)
		if(args != null && !args.isEmpty()) {
			String rawArgs = "";			
			
			for(String key : args.keySet())
				rawArgs += (key + "=" + URLEncoder.encode(args.get(key), "UTF-8") + "&");
			
			rawArgs = rawArgs.substring(0, rawArgs.length() - 1);
			
			try {
				url = new URI(this.url.toString() + "?" + rawArgs).toURL();
			} catch (MalformedURLException e) {
				e.printStackTrace();
			} catch (URISyntaxException e) {
				e.printStackTrace();
			}
		}
		
		// retry mechanism for unstable connections (increase robustness for temporary network issues)
		int connectErrors = 0;
		
		while(connectErrors < 3){
			try{
				System.out.println(method + " " + url.toString());
				
				// open HTTPS connection and configure request (set timeouts, headers, HTTP method and authentication)
				HttpsURLConnection conn = (HttpsURLConnection) url.openConnection();
				conn.setConnectTimeout(5000);
				conn.setDoInput(true);
				conn.setDoOutput(true);

				// only enable Output for methods that can send a body
				boolean sendsBody = (method == HTTPMethod.POST || method == HTTPMethod.PUT || method == HTTPMethod.PATCH);
				conn.setUseCaches(false);
				
				conn.setRequestProperty("User-Agent", "Mozilla/5.0");
				conn.setRequestProperty("Content-Type", "application/json");
				conn.setRequestProperty("Accept", "application/json");
				
				if (method == HTTPMethod.PATCH) {
					conn.setRequestMethod("POST");
					conn.setRequestProperty("X-HTTP-Method-Override", "PATCH");
				} else {
					conn.setRequestMethod(method.name());
				}
				// Basic Authentication (SN instance is accessed using a technical user)
				String token = user + ":" + pass;
				token = new String(Base64.getEncoder().encode(token.getBytes()));
				conn.setRequestProperty("Authorization", "Basic " + token);
				
				// write JSON payload for body-sending methods
				// POST creates records, PATCH/PUT updates records (e.g. worknotes, state)
				if (sendsBody) {
					OutputStreamWriter writer = new OutputStreamWriter(conn.getOutputStream(), StandardCharsets.UTF_8);
					writer.write(payload.toString());
					writer.close();
				} else {
					conn.connect();
				}
				
				// read response code and handle success/error responses
				responseCode = conn.getResponseCode();
				
				if (responseCode >= HttpsURLConnection.HTTP_BAD_REQUEST) {
					// error handling (read error stream and throw APIException containing status and message
					content = IOUtils.toByteArray(conn.getErrorStream());
					
					// SN often returns a JSON object with "message" or error details
					JSONObject errorJSON = new JSONObject(new String(content, StandardCharsets.UTF_8));
					throw new APIException(errorJSON.optString("message", "Unknown Error"), url, responseCode);
				} else {
					// success handling (read the response body (JSON))
					content = IOUtils.toByteArray(conn.getInputStream());
				}
				
				conn.disconnect();
				return;
			} catch(IOException e) {		// retry behavior on IO errors
				connectErrors++;
				if(connectErrors >= 3){
					throw e;
				}
			}
		}
	}
	
	// expose last HTTP status code (allows callers to inspect the result of the last request)
	public int getResponseCode() {
		return this.responseCode;
	}

	// parse response content as JSON 
	// executes the HTTP call (if not already done) and returns the response as JSON object
	public JSONObject getJSON() throws IOException, APIException {
		if(responseCode == -1) getContent();
		
		// if content is empty for some operations, return an empty JSON object instead of crashing
		if (content == null || content.length == 0) return new JSONObject();
		
		return new JSONObject(new String(content));
	}
}

package de.schoenn.ServiceNow4J.CMDB;

import org.json.JSONObject;
import de.schoenn.ServiceNow4J.TableEntry;

public class ConfigurationItem extends TableEntry {
	
	// SN-Filter: cmdb_ci.list
	
	// relevant CI fields for the project
	private String name;
	private String applicationSupportGroupSysID;
	private InstallStatus installStatus;
	private String location;
	private boolean isVirtual;
//	private String purpose;		// Erstmal nur eine Idee, könnte in die Description aufgenommen werden

	public ConfigurationItem(String sysID) {
		super(sysID);
		// TODO Auto-generated constructor stub
	}
	
	// getters
	public String getName(){
		return name;
	}

	public String getApplicationSupportGroupSysID() {
		return applicationSupportGroupSysID;
	}

	public InstallStatus getInstallStatus() {
		return installStatus;
	}

	public String getLocation() {
		return location;
	}

	public boolean isVirtual() {
		return isVirtual;
	}

	// setters
	public void setName(String name) {

	}

	public void setApplicationSupportGroupSysID(String applicationSupportGroup) {
		this.applicationSupportGroupSysID = applicationSupportGroup;
	}

	public void setInstallStatus(InstallStatus installStatus) {
		this.installStatus = installStatus;
	}

	public void setLocation(String location) {
		this.location = location;
	}

	public void setVirtual(boolean isVirtual) {
		this.isVirtual = isVirtual;
	}
	
	// JSON mapping:
	
	// populates this CI with values from a SN CMDB JSON record
	public void fromJSON(JSONObject result) {
		if (result == null) return;
		
		// simple string fields (depending on the instance, field names may differ)
		this.name = extractValueAsString(result, "name");
		this.location = extractReferenceDisplayValue(result, "location");
		this.isVirtual = extractReferenceBooleanValue(result, "virtual");
		
		// enum for install_status
		String rawStatus = result.optString("install_status", null);
		this.installStatus = InstallStatus.fromID(rawStatus);
		
		// reference field: application_support_group
		this.applicationSupportGroupSysID = extractReferenceSysID(result, "u_application_support_group");
	}
	
	// extracs the sys_id ("value") from a SN reference field
	// SN often returns a reference fields like: "application_support_group": { "link": "...", "value": "sysID" }
	private String extractReferenceSysID(JSONObject result, String fieldName) {
		if (!result.has(fieldName)) return null;
		
		Object raw = result.opt(fieldName);
		if (raw instanceof JSONObject) {
			return ((JSONObject) raw).optString("value", null);
		}
		
		// if it's already returned as a String
		if (raw instanceof String) {
			return (String) raw;
		}
		
		return null;
	}
	
	// extracts a readable string from a field that might be:
	// plain string OR a reference object with "display_value"/"value"
	// use this if the instance returns location as an object
	private String extractReferenceValueAsString(JSONObject result, String fieldName) {
		if (!result.has(fieldName)) return null;
		
		Object raw = result.opt(fieldName);
		
		if (raw instanceof JSONObject) {
			JSONObject obj = (JSONObject) raw;
			
			// if the instance is using "display_value", prefere that:
			String display = obj.optString("display_value", null);
			
			if (display != null && !display.isBlank()) return display;
			
			// otherwise fall back to the raw value:
			return obj.optString("value", null);
		}
		
		if (raw instanceof String) {
			return (String) raw;
		}
		
		return null;
	}
	
	private String extractReferenceDisplayValue(JSONObject result, String fieldName) {
		if (!result.has(fieldName)) return null;
		
		Object raw = result.opt(fieldName);
		
		if (raw instanceof JSONObject) {
			JSONObject obj = (JSONObject) raw;
			return obj.optString("display_value", null);
		}
		
		if (raw instanceof String) {
			return (String) raw;
		}
		
		return null;
	}
	
	private boolean extractReferenceBooleanValue(JSONObject result, String fieldName) {
		if (!result.has(fieldName)) return false;
		
		Object raw = result.opt(fieldName);
		
		if (raw instanceof JSONObject) {
			JSONObject obj = (JSONObject) raw;
			String value = obj.optString("value", "false");
			return Boolean.parseBoolean(value);
		}
		
		if (raw instanceof String) {
			return Boolean.parseBoolean((String) raw);
		}
		
		if (raw instanceof Boolean) {
			return (Boolean) raw;
		}
		
		return false;
	}
	
	private String extractValueAsString(JSONObject result, String fieldName) {
		if (!result.has(fieldName)) return null;
		
		Object raw = result.opt(fieldName);
		
		// case 1: SN returned object with value/display_value
		if (raw instanceof JSONObject) {
			JSONObject obj = (JSONObject) raw;
			return obj.optString("value", null);
		}
		
		// case 2: SN returned plain string
		if (raw instanceof String) {
			return (String) raw;
		}
		
		return null;
	}
	
	@Override
	public String toString() {
		return "ConfigurationItem{sysID=" + getSysID()
		+ ", name=" + name
		+ ", installStatus=" + installStatus
		+ ", location=" + location
		+ ", isVirtual=" + isVirtual
		+ "}";
	}
}

	// useful when reading CI data from the SN API
	public static InstallStatus fromID(String id) {
		if (id == null) return null;
		
		String cleanedID = id.trim();
		
		for (InstallStatus status : values()) {
			if (status.getID().equals(cleanedID)) {
				return status;
			}
		}
		
		return null;
	}

	public String toString() {
		return id;
	}
